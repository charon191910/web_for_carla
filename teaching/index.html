# 任务2: 设计新奖励函数
def custom_reward_function(obs, action, info):
    # 自定义奖励逻辑
    safety_reward = compute_safety_reward(obs)
    efficiency_reward = compute_efficiency_reward(obs)
    comfort_reward = compute_comfort_reward(action)

    # 创新性奖励
    innovation_reward = compute_innovation_reward(obs)

    total_reward = (0.3 * safety_reward + 
                    0.3 * efficiency_reward + 
                    0.2 * comfort_reward + 
                    0.2 * innovation_reward)

    return total_reward

def assess_student(sudent_id):
    assessments = {}
    for skill, levels in skills_assessment.items():
        level = evaluate_skill(sudent_id, skill)
        assessments[skill] = levels[level]
    return assessments
